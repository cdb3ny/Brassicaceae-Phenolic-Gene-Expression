grep AT5G07990 arabipsis_genes_names.txt > AT5G07990_ref_list.txt

#get just the locus name
cut -f1 AT5G07990_ref_list.txt > AT5G07990-loci.txt 

#recover the loci that are up or down regulated in any file, -f1 here recovers both the comparison name and the locus name. Converted so column 1 is the comparison and 2 is the locus name
grep -f AT5G07990-loci.txt * | cut -f1 | sed 's/:/\t/g' > AT5G07990-hits.txt

#now we want to go back and match comparison, locus, AT5G07990 name
#if field 1 in AT5G07990_ref_list.txt matches field 2 in AT5G07990-hits.txt,  recover the row in AT5G07990-hits.txt and add field 2 from AT5G07990_ref_list.txt to the recoverd row. 

awk -F'\t' '
NR==FNR {
    ref[$1] = $2
    next
}
($2 in ref) {
    print $0 "\t" ref[$2]
}
' AT5G07990_ref_list.txt AT5G07990-hits.txt > AT5G07990-matched.txt
